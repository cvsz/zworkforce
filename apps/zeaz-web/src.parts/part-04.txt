  try {
    const url = new URL(request.url);
    if (url.pathname.startsWith("/api/")) return api(request, env, ctx);

    return html(landing(), 404);
  } catch (error) { console.error("unhandled_request_error", error); return response("Internal Server Error", 500, { "content-type": "text/plain; charset=utf-8" }); }
} };
