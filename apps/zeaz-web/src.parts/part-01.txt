    const validation = validateLeadPayload(input);

    ctx.waitUntil(notify(env, validation.value, result.id));
    return json({ ok: true, requestId: result.id, message: validation.value.locale === "th" ? "เราได้รับคำขอของคุณแล้ว" : "Your request has been received." }, 201);
  }
  if (request.method === "GET" && url.pathname === "/api/admin/early-access") {
    if (!env.ADMIN_API_TOKEN || request.headers.get("authorization") !== `Bearer ${env.ADMIN_API_TOKEN}`) return json({ error: { code: "unauthorized", message: "Unauthorized." } }, 401);
    const limit = Math.min(Math.max(Number(url.searchParams.get("limit") || 50), 1), 200);
    const status = clean(url.searchParams.get("status"), 20) || "new";
    const rows = await env.DB.prepare("SELECT id,created_at,updated_at,email,full_name,company,role,country,interest,message,locale,source,status,marketing_consent,cf_country FROM early_access_requests WHERE status=?1 ORDER BY created_at DESC LIMIT ?2").bind(status, limit).all();

  }
